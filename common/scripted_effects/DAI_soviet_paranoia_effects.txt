DAI_SOV_historical_paranoia_daily_update = {
	# New games need no cooldown. Pre-controller saves lack usable timing history.
	if = {
		limit = {
			is_ai = yes
			DAI_SOV_is_historical_ai = yes
			NOT = { has_country_flag = DAI_SOV_paranoia_controller_v3_initialized }
		}
		if = {
			limit = { NOT = { has_country_flag = DAI_SOV_paranoia_controller_initialized } }
			if = {
				limit = { has_country_flag = SOV_paranoia_system_active_flag }
				set_country_flag = { flag = DAI_SOV_forge_production_reports_cooldown value = 1 days = 180 }
				set_country_flag = { flag = DAI_SOV_inspection_in_the_navy_cooldown value = 1 days = 180 }
				set_country_flag = { flag = DAI_SOV_inspection_in_the_airforce_cooldown value = 1 days = 180 }
			}
			set_country_flag = DAI_SOV_paranoia_controller_initialized
		}
		else = {
			# Old mirrors began at launch. Append the missing active period once they expire.
			set_country_flag = DAI_SOV_forge_production_reports_legacy_cooldown_pending
			set_country_flag = DAI_SOV_inspection_in_the_navy_legacy_cooldown_pending
			set_country_flag = DAI_SOV_inspection_in_the_airforce_legacy_cooldown_pending
		}
		set_country_flag = DAI_SOV_paranoia_controller_v3_initialized
	}

	if = {
		limit = {
			has_country_flag = DAI_SOV_forge_production_reports_legacy_cooldown_pending
			NOT = { has_country_flag = DAI_SOV_forge_production_reports_cooldown }
		}
		set_country_flag = { flag = DAI_SOV_forge_production_reports_cooldown value = 1 days = 90 }
		clr_country_flag = DAI_SOV_forge_production_reports_legacy_cooldown_pending
	}

	if = {
		limit = {
			has_country_flag = DAI_SOV_inspection_in_the_navy_legacy_cooldown_pending
			NOT = { has_country_flag = DAI_SOV_inspection_in_the_navy_cooldown }
		}
		set_country_flag = { flag = DAI_SOV_inspection_in_the_navy_cooldown value = 1 days = 7 }
		clr_country_flag = DAI_SOV_inspection_in_the_navy_legacy_cooldown_pending
	}

	if = {
		limit = {
			has_country_flag = DAI_SOV_inspection_in_the_airforce_legacy_cooldown_pending
			NOT = { has_country_flag = DAI_SOV_inspection_in_the_airforce_cooldown }
		}
		set_country_flag = { flag = DAI_SOV_inspection_in_the_airforce_cooldown value = 1 days = 7 }
		clr_country_flag = DAI_SOV_inspection_in_the_airforce_legacy_cooldown_pending
	}

	if = {
		limit = {
			is_ai = yes
			DAI_SOV_is_historical_ai = yes
			has_country_flag = SOV_paranoia_system_active_flag
			check_variable = { DAI_SOV_paranoia_pp_bank > 24 }
			DAI_SOV_should_wait_for_paranoia_reducing_focus = no
		}

		if = {
			limit = {
				check_variable = { SOV_paranoia > 19 }
				NOT = { has_decision = SOV_forge_production_reports }
				NOT = { has_country_flag = DAI_SOV_forge_production_reports_cooldown }
			}
			activate_decision = SOV_forge_production_reports
			subtract_from_variable = { DAI_SOV_paranoia_pp_bank = 25 }
		}
		else_if = {
			limit = {
				check_variable = { SOV_paranoia > 20 }
				check_variable = { DAI_SOV_paranoia_pp_bank > 49 }
				NOT = { has_country_flag = SOV_paranoia_conducting_inspection_flag }
				NOT = { has_decision = SOV_inspection_in_the_navy }
				NOT = { has_decision = SOV_inspection_in_the_airforce }
				NOT = { has_country_flag = DAI_SOV_inspection_in_the_navy_cooldown }
				SOV_is_stalinist = yes
				OR = {
					any_navy_leader = {
						SOV_character_can_be_purged = yes
					}
					NOT = { has_idea = SOV_anti_soviet_military_thinking_banned_navy }
					NOT = { has_idea = SOV_shipyard_staff_purged }
				}
			}
			activate_decision = SOV_inspection_in_the_navy
			subtract_from_variable = { DAI_SOV_paranoia_pp_bank = 50 }
		}
		else_if = {
			limit = {
				check_variable = { SOV_paranoia > 20 }
				check_variable = { DAI_SOV_paranoia_pp_bank > 49 }
				NOT = { has_country_flag = SOV_paranoia_conducting_inspection_flag }
				NOT = { has_decision = SOV_inspection_in_the_navy }
				NOT = { has_decision = SOV_inspection_in_the_airforce }
				NOT = { has_country_flag = DAI_SOV_inspection_in_the_airforce_cooldown }
				SOV_is_stalinist = yes
				OR = {
					NOT = { has_idea = SOV_anti_soviet_military_thinking_banned_airforce }
					NOT = { has_idea = SOV_ground_based_training }
				}
			}
			activate_decision = SOV_inspection_in_the_airforce
			subtract_from_variable = { DAI_SOV_paranoia_pp_bank = 50 }
		}
	}
}
