# An active regional bonus normally holds the sequence. Once that region is
# nearly full, the next decision may start while the original bonus continues.
DAI_GER_reich_labor_state_development_bonus_blocks_next_decision = {
	OR = {
		AND = {
			has_country_flag = DAI_GER_western_state_development_active
			NOT = { DAI_GER_western_state_development_area_nearly_full = yes }
		}
		AND = {
			has_country_flag = DAI_GER_southern_state_development_active
			NOT = { DAI_GER_southern_state_development_area_nearly_full = yes }
		}
		has_country_flag = DAI_GER_eastern_state_development_active
		AND = {
			has_country_flag = DAI_GER_central_state_development_active
			NOT = { DAI_GER_central_state_development_area_nearly_full = yes }
		}
		AND = {
			has_country_flag = DAI_GER_northern_state_development_active
			NOT = { DAI_GER_northern_state_development_area_nearly_full = yes }
		}
	}
}
